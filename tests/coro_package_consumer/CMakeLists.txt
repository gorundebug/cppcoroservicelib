cmake_minimum_required(VERSION 3.24)
project(coro_package_consumer LANGUAGES CXX)

find_package(cppcoroservicelib CONFIG REQUIRED)
get_target_property(dns_dependencies servicelib::coro_dns INTERFACE_LINK_LIBRARIES)
if(NOT "Boost::headers" IN_LIST dns_dependencies)
  message(FATAL_ERROR "Installed Coro DNS target is missing its Boost headers dependency")
endif()
get_target_property(runtime_backend servicelib::coro_io INTERFACE_SERVICELIB_CORO_IO_BACKEND)
if(NOT runtime_backend STREQUAL EXPECTED_IO_BACKEND)
  message(FATAL_ERROR "Coro I/O backend mismatch: consumer=${EXPECTED_IO_BACKEND}, runtime=${runtime_backend}")
endif()
set(EXPECTED_IO_BACKEND "epoll" CACHE STRING "Expected installed I/O backend")
if(NOT EXPECTED_IO_BACKEND MATCHES "^(epoll|uring)$")
  message(FATAL_ERROR "EXPECTED_IO_BACKEND must be epoll or uring")
endif()
add_executable(coro_package_consumer main.cpp)
target_link_libraries(coro_package_consumer PRIVATE servicelib::grpc)
target_compile_definitions(coro_package_consumer PRIVATE
    EXPECTED_CORO_URING=$<STREQUAL:${EXPECTED_IO_BACKEND},uring>)
enable_testing()
add_test(NAME installed_coro_backend COMMAND coro_package_consumer)
set_tests_properties(installed_coro_backend PROPERTIES TIMEOUT 15)
